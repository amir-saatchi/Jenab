    use: llm.select

  - id: news_rows
    use: transform.map
    with:
      items: ${{ steps.top5.items }}
      fields:
        url: ${{ item.url }}
        title: ${{ item.title }}
        reason: ${{ item.reason }}
        date: ${{ today() }}

  - id: save_news
    use: db.insert_many
    with:
      table: btc_news
      on_conflict: ignore
      rows: ${{ steps.news_rows.items }}
